Java Program — Household Waste Analyzer

Program Name: Household Waste Analyzer & Tracker Game

Description: A Java console application that stores 7-day waste weights in a double[] array, calculates totals and averages, identifies peak days, sorts weights in descending order, and gives real-time High/Low warning feedback.

import java.util.Scanner;
import java.util.Arrays;

public class HouseholdWasteAnalyzer {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        double[] weights = new double[7];  
        String[] days = {"Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday", "Sunday"};  
        double total = 0;  
          
        System.out.println("=== HOUSEHOLD WASTE ANALYZER & TRACKER GAME ===");  
          
        System.out.print("Set your Daily High Waste Limit (kg): ");  
        double warningLimit = scanner.nextDouble();  
        System.out.println("--------------------------------------------------");  
        System.out.println("Game started! Tracking daily waste for 7 days...\n");  
          
        for (int i = 0; i < 7; i++) {  
            while (true) {  
                System.out.print("Enter waste weight for " + days[i] + " (kg): ");  
                double weight = scanner.nextDouble();  
                  
                if (weight >= 0) {  
                    weights[i] = weight;  
                    total += weight;  
                      
                    if (weight > warningLimit) {  
                        System.out.println("   --> ⚠️ HIGH WASTE ALERT! (" + weight + " kg exceeds limit)");  
                    } else {  
                        System.out.println("   --> ✅ LOW WASTE! Good eco-friendly day.");  
                    }  
                    System.out.println();  
                    break;  
                } else {  
                    System.out.println("Invalid input! Please enter a nonnegative value.");  
                }  
            }  
        }  
          
        double average = total / 7;  
          
        double maxWeight = weights[0];  
        int maxIndex = 0;  
        for (int i = 1; i < weights.length; i++) {  
            if (weights[i] > maxWeight) {  
                maxWeight = weights[i];  
                maxIndex = i;  
            }  
        }  
          
        System.out.println("================ RESULT SUMMARY ================");  
        System.out.printf("Total Waste Produced: %.2f kg\n", total);  
        System.out.printf("Average Daily Weight: %.2f kg\n", average);  
        System.out.println("Peak Waste Day: " + days[maxIndex] + " (" + maxWeight + " kg)");  
          
        double[] sortedWeights = weights.clone();  
        Arrays.sort(sortedWeights);  
          
        System.out.println("\n----------- WASTE WEIGHTS (DESCENDING) -----------");  
        for (int i = sortedWeights.length - 1; i >= 0; i--) {  
            System.out.printf("%.2f kg\n", sortedWeights[i]);  
        }  
        System.out.println("==================================================");  
          
        scanner.close();  
    }
}
